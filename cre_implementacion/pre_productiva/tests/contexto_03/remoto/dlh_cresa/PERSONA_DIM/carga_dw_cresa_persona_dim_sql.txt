CREATE OR REPLACE TEMPORARY VIEW tmp_telefono
as 
select CNUMEROIDENTIFICACION,
CNUMERO ,
max(DFECHAINGRESO) max
from parquet.`/Volumes/dlh_cresa/bronze/vol_bronze_siac/genvtclientedirecciontelef_cresa/`
where CCODIGOTIPOTELEFONO='002'
and CDESCRIPCION='Domicilio'
group by CNUMEROIDENTIFICACION,
CNUMERO;


CREATE OR REPLACE TEMPORARY VIEW tmp_persona
as 
select  a.ICODIGOPERSONA ,MAX(a.ISECUENCIA)ISECUENCIA, MAX(a.DFECHAINGRESO) DFECHAINGRESO
 from parquet.`/Volumes/dlh_cresa/bronze/vol_bronze_siac/genmtelefono_cresa/` a
 where a.CCODIGOTIPOTELEFONO='002'
 group by a.ICODIGOPERSONA;

CREATE OR REPLACE TEMPORARY VIEW tmp_persona1
as 
 select a.ICODIGOPERSONA,a.DFECHAINGRESO,b.CNUMERO from tmp_persona a
 inner join parquet.`/Volumes/dlh_cresa/bronze/vol_bronze_siac/genmtelefono_cresa/`  b
 on a.ICODIGOPERSONA =b.ICODIGOPERSONA
 and a.ISECUENCIA=b.ISECUENCIA
 and a.DFECHAINGRESO=b.DFECHAINGRESO;


CREATE OR REPLACE table dlh_cresa.staging.tmp_telefonos_cresa
as 
select  b.CNUMEROIDENTIFICACION,a.CNUMERO 
from tmp_persona1 a
inner join parquet.`/Volumes/dlh_cresa/bronze/vol_bronze_siac/genmpersona_cresa/` b
on a.ICODIGOPERSONA =b.ICODIGOPERSONA 
 group by b.CNUMEROIDENTIFICACION,a.CNUMERO;
 
CREATE OR REPLACE temporary view tmp_telefono2
as 
 select a.CNUMEROIDENTIFICACION,
a.CNUMERO from tmp_telefono a
where not EXISTS (select 1 from dlh_cresa.staging.tmp_telefonos_cresa  b
where a.CNUMEROIDENTIFICACION=b.CNUMEROIDENTIFICACION);

insert into dlh_cresa.staging.tmp_telefonos_cresa
select * from tmp_telefono2;


MERGE INTO dlh_cresa.gold.dw_cresa_persona_dim  AS target
USING  dlh_cresa.staging.tmp_telefonos_cresa AS source
ON target.num_identificacion = source.CNUMEROIDENTIFICACION
WHEN MATCHED THEN
UPDATE SET target.des_telefono2 = source.CNUMERO;
