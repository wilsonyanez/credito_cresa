TRUNCATE TABLE dlh_cresa.staging.temp_persona_dim;  

INSERT INTO dlh_cresa.staging.temp_persona_dim
select 
  AddressCity,
  AddressCountryRegionId,
  AddressCountryRegionISOCode,
  AddressDescription,
  AddressLocationId,
  AddressLocationRoles,
  AddressRecordId,
  LPAD(CAST(AddressState AS STRING), 2, '0') AddressState,
  AddressStreet,
  AddressStreetNumber,
  AddressValidFrom,
  AddressValidTo,
  CreditLimit,
  CredManCreditLimitDate,
  CredManCreditLimitExpiryDate,
  CredManCustomerSince,
  CredManEligibleCreditMax,
  CredManCustCreditMaxAlt,
  CredManGroupId,
  CustomerAccount,
  CustomerGroupId,
  DeliveryAddressDistrictName,
  DeliveryAddressState,
  DeliveryAddressStreet,
  -- DeliveryAddressStreetNumber,
  TRY_CAST(NULLIF(DeliveryAddressValidFrom, '') AS TIMESTAMP) AS DeliveryAddressValidFrom,
  TRY_CAST(NULLIF(DeliveryAddressValidTo, '') AS TIMESTAMP) AS DeliveryAddressValidTo,
  DeliveryAddressZipCode,
  ElectronicLocationId,
  FullPrimaryAddress,
  InvoiceAddressCity,
  InvoiceAddress,
  InvoiceAddressCountryRegionISOCode,
  InvoiceAddressLocationId,
  InvoiceAddressState,
  InvoiceAddressStreetNumber,
  TRY_CAST(NULLIF(InvoiceAddressValidFrom, '') AS TIMESTAMP) AS InvoiceAddressValidFrom,
  TRY_CAST(NULLIF(InvoiceAddressValidTo, '') AS TIMESTAMP) AS InvoiceAddressValidTo,
  InvoiceAddressZipCode,
  IsElectronicInvoice,
  -- IsFinalUser,
  -- IsServiceDeliveryAddressBased,
  LanguageId,
  NameAlias,
  NatureOfAssessee,
  OrganizationName,
  PartyNumber,
  PaymentMethod,
  PaymentTerms,
  TRY_CAST(NULLIF(PersonBirthDay, '') AS INT) AS PersonBirthDay,
  PersonBirthMonth,
  TRY_CAST(NULLIF(PersonBirthYear, '') AS INT) AS PersonBirthYear,
  PersonFirstName,
  PersonLastName,
  PersonMiddleName,
  PrimaryContactEmailDescription,
  PrimaryContactEmailIsIM,
  -- PrimaryContactEmailPurpose,
  -- PrimaryContactEmailRecordId,
  PrimaryContactPhone,
  PrimaryContactPhoneDescription,
  PrimaryContactPhoneExtension,
  PrimaryContactPhonePurpose,
  TRY_CAST(NULLIF(PrimaryContactPhoneRecordId, '') AS BIGINT) AS PrimaryContactPhoneRecordId,
  PrimaryContactTelex,
  SalesAccountNumber,
  SalesCurrencyCode,
  SalesTaxGroup,
  SiteId,
  VATNum,
  CredManExclude,
  InvoiceAddressCountryRegionId,
  PartyType,
  '0001' dataAreaId,
  DefaultDimensionDisplayValue,
  -- PrimaryContactTelexPurpose,
  CASE 
    WHEN PersonBirthDay = '' OR PersonBirthMonth = '' OR PersonBirthYear = '' 
         OR PersonBirthDay IS NULL OR PersonBirthMonth IS NULL OR PersonBirthYear IS NULL
    THEN NULL
    ELSE TO_TIMESTAMP(
      CONCAT(
        LPAD(PersonBirthDay, 2, '0'), '-', 
        CASE LOWER(PersonBirthMonth)
          WHEN 'january' THEN '01'
          WHEN 'february' THEN '02'
          WHEN 'march' THEN '03'
          WHEN 'april' THEN '04'
          WHEN 'may' THEN '05'
          WHEN 'june' THEN '06'
          WHEN 'july' THEN '07'
          WHEN 'august' THEN '08'
          WHEN 'september' THEN '09'
          WHEN 'october' THEN '10'
          WHEN 'november' THEN '11'
          WHEN 'december' THEN '12'
          ELSE NULL
        END,
        '-', 
        PersonBirthYear,
        ' 00:00:00'
      ),
      'dd-MM-yyyy HH:mm:ss'
    )
  END AS fecha_nacimiento,
  primarycontactemail
from carga_temp_persona_parquet;
