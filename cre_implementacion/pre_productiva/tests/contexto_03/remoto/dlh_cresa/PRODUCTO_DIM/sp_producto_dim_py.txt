import requests

# Función para enviar un mensaje usando la API personalizada
def send_telegram_message(resultado_ejecucion, detalle_ejecucion):
    # Ahora solo enviamos los parámetros 'ResultadoEjecucion' y 'DetalleEjecucion'
    url = f"http://200.7.249.20/api/v1/insertar_log?CodigoProceso=11&ResultadoEjecucion={resultado_ejecucion}&DetalleEjecucion={detalle_ejecucion}"
    
    response = requests.get(url)
    if response.status_code != 200:
        print(f"Error enviando mensaje a la API: {response.text}")
