MERGE INTO dlh_cresa.gold.dw_cresa_producto_dim p
USING dlh_cresa.staging.temp_marca_dim m
ON  p.id_empresa = m.id_empresa -- AND p.cod_marca = m.cod_marca
and upper(p.cod_marca)  = UPPER(m.cod_marca)
-- or p.cod_marca  =INITCAP(m.cod_marca)
WHEN MATCHED THEN
  UPDATE SET
    p.des_marca = m.des_marca;

MERGE INTO dlh_cresa.gold.dw_cresa_producto_dim p
USING dlh_cresa.staging.temp_linea_dim l
ON p.id_empresa = l.id_empresa AND p.cod_linea = l.cod_linea
WHEN MATCHED THEN
  UPDATE SET
    p.des_linea = l.des_linea;


MERGE INTO dlh_cresa.gold.dw_cresa_producto_dim p
USING dlh_cresa.staging.temp_grupo_dim g
ON p.id_empresa = g.id_empresa AND p.cod_grupo = g.cod_grupo
WHEN MATCHED THEN
  UPDATE SET
    p.des_grupo = g.des_grupo;


MERGE INTO dlh_cresa.gold.dw_cresa_producto_dim p
USING dlh_cresa.staging.temp_subgrupo_dim sg
ON p.id_empresa = sg.id_empresa AND p.cod_subgrupo = sg.cod_subgrupo
WHEN MATCHED THEN
  UPDATE SET
    p.des_subgrupo = sg.des_subgrupo;


MERGE INTO dlh_cresa.gold.dw_cresa_producto_dim p
USING dlh_cresa.staging.temp_categoria_producto_dim c
ON p.id_empresa = c.id_empresa AND p.cod_categoria = c.cod_categoria
WHEN MATCHED THEN
  UPDATE SET
    p.des_categoria = c.des_categoria;

MERGE INTO dlh_cresa.gold.dw_cresa_producto_dim p
USING dlh_cresa.staging.temp_ciclo_producto_dim c
ON p.cod_estado_producto = c.LifecycleStateId --p.id_empresa = c.id_empresa
WHEN MATCHED THEN
  UPDATE SET
    p.des_estado_producto = c.LifecycleStateId;

MERGE INTO dlh_cresa.gold.dw_cresa_producto_dim p
USING dlh_cresa.staging.temp_capacidad_dim sg
ON p.id_empresa = sg.id_empresa AND p.cod_capacidad = sg.cod_capacidad
WHEN MATCHED THEN
  UPDATE SET
    p.des_capacidad = sg.des_capacidad;

MERGE INTO dlh_cresa.gold.dw_cresa_producto_dim p
USING dlh_cresa.staging.temp_proveedores_mktpl_dim sg
ON  p.cod_producto = sg.cod_producto
WHEN MATCHED THEN
  UPDATE SET
    p.ruc_proveedor= sg.ruc_proveedor, 
    p.codigo_proveedor= sg.codigo_proveedor , 
    p.razon_social_proveedor = sg.razon_social_proveedor;
